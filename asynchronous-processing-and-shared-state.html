    6. Asynchronous Processing and Shared State

    6.1. Example 19: Future, Stream, and Timers

    Many things happen “later”: timer ticks, keystrokes, a server’s response. [Dart] has two types for this, at the core of the language:

    • [Future<T>]: UNE value, later (a server’s response, a choice made in a window);
    • [Stream<T>]: a sequence of SUITE values over time (ticks, messages, GPS positions...).

    [Flutter] has a widget to display each one, rebuilding it when a value arrives: [FutureBuilder] and [StreamBuilder]. And for what must happen after a delay: [Timer].

    // =========================================================================
// lib/ex19_asynchrone/main.dart - Future, Stream, Timers
// -------------------------------------------------------------------------
// Many things happen “later”: the ticks of a timer,
// keyboard keystrokes, a server’s response. Dart has two types for
// this, at the core of the language:
//    - Future<T>: a value, later (a server’s response);
//    - Stream<T>: a SUITE of values, over time (ticks,
//      messages...).
// Flutter has a widget to display each one, rebuilding it when a
// value arrives: FutureBuilder and StreamBuilder.
// And for what needs to happen after a delay: Timer (dart:async).
// =========================================================================

import 'dart:async';
import 'dart:math';

import 'package:flutter/material.dart';

const medecins = ['Marie PELISSIER', 'Paul TATOU', 'Sophie DUPONT', 'Jacques DURAND', 'Martine PELLETIER'];
final _hasard = Random();

// A simulated “server”: the response (a Future) arrives after a random delay
Future<List<String>> rechercher(String texte) async {
  await Future<void>.delayed(Duration(milliseconds: 200 + _hasard.nextInt(600)));
  return medecins.where((m) => m.toLowerCase().contains(texte)).toList();
}

// 1. A Stream created by an “async*” function: each yield ENVOIE returns a
// value to those listening to the Stream
Stream<int> compteARebours(int depart) async* {
  for (var reste = depart; reste >= 0; reste--) {
    yield reste;
    if (reste > 0) await Future<void>.delayed(const Duration(seconds: 1));
  }
}

String heureCourante() => DateTime.now().toString().substring(11, 19);

void main() {
  runApp(
    MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const PageAsynchrone(),
    ),
  );
}

class PageAsynchrone extends StatefulWidget {
  const PageAsynchrone({super.key});

  @override
  State<PageAsynchrone> createState() => _PageAsynchroneState();
}

class _PageAsynchroneState extends State<PageAsynchrone> {
  final List<String> _journal = [];
   // The stream and the Future are created UNE times (and not on every build:
   // otherwise, every build call would restart the countdown!)
  final Stream<int> _rebours = compteARebours(20);
   // 3. A Future that completes after one second
  final Future<String> _message = Future.delayed(const Duration(seconds: 1), () => 'trois (une seconde plus tard)');

   // 2. search as you type
  Timer? _antiRebond;
  int _numeroRequete = 0; // the number of the DERNIÈRE search launched
  List<String> _resultats = [];

  void _tracer(String message) => setState(() => _journal.add('${heureCourante()} $message'));

   // called with each keystroke
  void _saisie(String texte) {
     // ANTI-REBOND: Each keystroke resets the timer for the
     // previous one; the search only starts after 400 ms without keystrokes
    _antiRebond?.cancel();
    _antiRebond = Timer(const Duration(milliseconds: 400), () => _lancer(texte.trim().toLowerCase()));
  }

  Future<void> _lancer(String texte) async {
     // fewer than two letters: no search, no results
    if (texte.length < 2) {
      _numeroRequete++; // a response still pending will be ignored
      setState(() => _resultats = []);
      return;
    }
     // "outdated": a response that arrives APRÈS after another search has been launched
     // search must be ignored (otherwise the slow response from an old
     // search would overwrite that of the new one)
    final numero = ++_numeroRequete;
    _tracer('recherche envoyée : « $texte »');
    final trouves = await rechercher(texte);
    if (!mounted) return;
    if (numero != _numeroRequete) {
      _tracer('réponse ignorée (périmée) : « $texte »');
      return;
    }
    setState(() => _resultats = trouves);
  }

  @override
  void dispose() {
    _antiRebond?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final styleTitre = Theme.of(context).textTheme.titleSmall;
    return Scaffold(
      appBar: AppBar(title: const Text("Exemple 19 - l'asynchrone")),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          Card(
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text('Un compte à rebours (un Stream et StreamBuilder)', style: styleTitre),
                   // StreamBuilder: reconstructs its content with each value from the stream;
                   // etat.data: the last value received (null before the first)
                  StreamBuilder<int>(
                    stream: _rebours,
                    builder: (context, etat) =>
                        Text(etat.hasData ? '${etat.data}' : '...', style: Theme.of(context).textTheme.displaySmall),
                  ),
                ],
              ),
            ),
          ),
          Card(
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text('Une recherche au fil de la frappe (anti-rebond, réponses périmées)', style: styleTitre),
                  TextField(
                    decoration: const InputDecoration(hintText: 'au moins deux lettres : pel, dur...'),
                    onChanged: _saisie,
                  ),
                  for (final r in _resultats) Text('• $r'),
                  if (_resultats.isEmpty) const Text('aucun résultat', style: TextStyle(color: Colors.grey)),
                ],
              ),
            ),
          ),
          Card(
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text('Attendre un Future (FutureBuilder)', style: styleTitre),
                   // as long as the Future is not complete: the pending content
                  FutureBuilder<String>(
                    future: _message,
                    builder: (context, etat) => switch (etat.connectionState) {
                      ConnectionState.done => Text('reçu : ${etat.data}'),
                      _ => const Text('en attente...', style: TextStyle(color: Colors.grey)),
                    },
                  ),
                ],
              ),
            ),
          ),
          Text('Journal', style: styleTitre),
          for (final (i, ligne) in _journal.indexed) Text('${i + 1}. $ligne', style: const TextStyle(fontSize: 12)),
        ],
      ),
    );
  }
}
    • lines 24–26: a simulated “server”: a function [async], whose response (a [Future]) arrives after a random delay. [await Future<void>.delayed(...)] waits for this delay without blocking the application;
    • lines 31–34: a [Stream] generated by a function [async*] (with an asterisk): an asynchronous generator. Each [yield] sends a value to those listening to [Stream]; between values, the function waits one second;
    • lines 61–63: the [Stream] and the [Future] are created UNE times, in fields of the [State] object, and not in [build]: Each rebuild would restart the countdown! This is the most common error with [FutureBuilder] and [StreamBuilder];
    • lines 66–68: the status of the search as the user types: a timer, the number of the last search initiated, and the results;
    • lines 73–77: debouncing: the program waits for the user to pause before querying the server, rather than sending a request with every keystroke. Each keystroke resets the timer from the previous keystroke ([cancel()]) and starts a new one: the search is only triggered after 400 milliseconds without a keystroke;
    • Lines 80–98: the search. An “out-of-date” response (one that arrives—APRÈS—after another search has been launched) must be ignored: otherwise, the slow response from an old search would override that of the new one. Each search is assigned a number ([++_numeroRequete]); when its response arrives, if it is no longer the number of the most recent search, it is ignored. [if (!mounted) return;]: the screen may have disappeared while waiting;
    • lines 102–104: the timer is stopped if the screen disappears;
    • lines 124–127: [StreamBuilder] listens for [Stream] and reconstructs its content with each new value. [builder] receives a snapshot ([etat], a [AsyncSnapshot]): [etat.hasData], [etat.data] (the last value received), [etat.hasError], [etat.connectionState]. When the widget disappears, [StreamBuilder] unsubscribes on its own;
    • lines 158–162: [FutureBuilder] waits for [Future]: until it is finished, the placeholder content; then, its value. [switch] on [etat.connectionState] chooses one or the other.

    We type “p,” “pe,” “pel” quickly: only one search is sent. Then “du,” followed by “dup” before the response: the response to “du” arrives too late, and it is ignored if “dup” has already received its response:

    In [JavaScript], the same thing is written using promises and timers; [Angular] adds the "observables" from [RxJS]. In [Dart], [Future] and [Stream] are in the base library, and [async], [await], [async*], [yield], and [await for] are in the language: this is why [Flutter] does not require a library for asynchronous operations.

    6.2. Example 20: A Shared State

    Same question as in previous lessons: what are the “scopes” for an application’s data?

    • A widget’s local state disappears when the widget is removed;
    • the state of a “store” is shared by all widgets and survives their removal… but not the application’s shutdown;
    • State stored on the device survives when the application is shut down.

    // =========================================================================
// lib/ex20_etat_partage/stores/panier.dart - a ÉTAT PARTAGÉ, with a provider
// -------------------------------------------------------------------------
// A “store” is a state that consists of HORS screens, in a single
// instance, shared by everyone who reads it. Example 14 did this
// manually (InheritedNotifier); the provider package, the most commonly used
// with Flutter (and recommended by its documentation), does the same thing
// in a single line:
//
//      ChangeNotifierProvider(create: (_) => Cart(), child: ...)
//
// The store is a regular ChangeNotifier. Standard structure:
//    - the ÉTAT: PRIVÉES data (_articles);
//    - ACTIONS: methods that modify the state and end with
//      with notifyListeners();
//    - calculated values: getters.
// A widget accesses it through its context:
//    - context.watch<Shopping  Cart>() : read ET, subscribe (rebuilt with every change);
//    - context.read<Shopping   Cart>() : read without subscribing (to call an action);
//    - context.select<Cart, int>((p) => p.nombre): subscribe to the value of UNE:
//      reconstructed only if the value of CETTE changes.
// The store lives as long as its supplier lives: it survives the removal of a
// screen... but not when the application stops.
// =========================================================================

import 'package:flutter/foundation.dart';

typedef Article = ({String libelle, double prix});

class Panier extends ChangeNotifier {
  final List<Article> _articles = [];

   // A view NON MODIFIABLE of the list: only actions change it
  List<Article> get articles => List.unmodifiable(_articles);
  int get nombre => _articles.length;
  double get total => _articles.fold(0, (s, a) => s + a.prix);

  void ajouter(Article article) {
    _articles.add(article);
    notifyListeners();
  }

  void vider() {
    _articles.clear();
    notifyListeners();
  }
}

    A store is a state that spans multiple screens, existing as a single instance shared by all who access it. Example 14 implemented this manually. The [provider] package, the most commonly used one along with [Flutter] (and the one recommended in its documentation), does the same thing in a single line: [ChangeNotifierProvider(create: (_) => Panier(), child: ...)].

    • Line 31: the state: private data;
    • lines 34–36: calculated values: getters. [List.unmodifiable]: a read-only view of the list; only actions can modify it;
    • lines 38–45: the actions: methods—the only ones that modify the state—which end with [notifyListeners()].

    // =========================================================================
// lib/ex20_etat_partage/stores/preferences.dart - a state that survives shutdown
// -------------------------------------------------------------------------
// For data to survive a shutdown of the application, it must be saved
// SUR The APPAREIL. The shared_preferences package provides a small dictionary
// key -> value (bool, int, double, String), specific to the app and
// stored on the device (on the web: in the browser's localStorage).
//    - at startup, the stored values are read (main.dart);
//    - they are rewritten with every change.
// No sensitive data (passwords, tokens) is stored here: they
// are stored in plain text.
// =========================================================================

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';

class Preferences extends ChangeNotifier {
  static const _cleSombre = 'exemple20.sombre';
  static const _cleTaille = 'exemple20.taille';

  final SharedPreferences _stockage;
  bool _sombre;
  double _taille; // text size

   // the initial state: what the device has stored; otherwise, the default values
  Preferences(this._stockage)
    : _sombre = _stockage.getBool(_cleSombre) ?? false,
      _taille = _stockage.getDouble(_cleTaille) ?? 16;

  bool get sombre => _sombre;
  double get taille => _taille;

   // a parameter designated as optional by preference: only the provided value is changed
  void modifier({bool? sombre, double? taille}) {
    if (sombre != null) {
      _sombre = sombre;
      _stockage.setBool(_cleSombre, sombre); // text on the device
    }
    if (taille != null) {
      _taille = taille;
      _stockage.setDouble(_cleTaille, taille);
    }
    notifyListeners();
  }
}

    For data to survive after the application is shut down, it must be stored on the device. The [shared_preferences] package provides a small key->value dictionary ([bool], [int], [double], [String]), specific to the application and stored on the device (on the web: in the browser’s [localStorage]).

    • Line 21: the storage data, received upon creation;
    • lines 26–28: the initial state: what the device has stored, or default values otherwise. The fields are initialized by the manufacturer’s “initialization list” (after the colon), before the body;
    • Lines 34–43: a single action, with optional named parameters: only the provided values are changed, and they are immediately written to the device ([setBool], [setDouble]).

    Do not store any sensitive data (passwords, tokens) such as JAMAIS there: such data is written in plain text. For this purpose, there is the [flutter_secure_storage] package, which uses the system’s vault ([Keystore] from [Android], [Keychain] from [iOS]).

// lib/ex20_etat_partage/main.dart - three "ranges" of data
// -------------------------------------------------------------------------
//    - the state LOCAL of a widget (CompteurLocal) disappears along with it;
//    - the state of a MAGASIN (the shopping cart) is shared by all widgets
//      and survives their removal—but not when the application is shut down;
//    - the state stored on the device (preferences) survives shutdown.
// And Visibility(maintainState: true): the content is CACHÉ, but not
// removed; its local state is preserved.
// Uncheck and then recheck "Show Field," then stop and restart
// the app: compare the results.
// =========================================================================

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'package:shared_preferences/shared_preferences.dart';

import 'composants/boutique.dart';
import 'composants/compteur_local.dart';
import 'composants/resume_panier.dart';
import 'magasins/panier.dart';
import 'magasins/preferences.dart';

Future<void> main() async {
   // Essential before using a "plugin" (Android-specific code, iOS,
   // web...) such as shared_preferences, AVANT, runApp
  WidgetsFlutterBinding.ensureInitialized();
  final stockage = await SharedPreferences.getInstance();
  runApp(
     // MultiProvider: multiple providers, nested within one another,
     // above the entire application
    MultiProvider(
      providers: [
        ChangeNotifierProvider(create: (_) => Panier()),
        ChangeNotifierProvider(create: (_) => Preferences(stockage)),
      ],
      child: const Application(),
    ),
  );
}

class Application extends StatelessWidget {
  const Application({super.key});

  @override
  Widget build(BuildContext context) {
     // watch: the app is rebuilt when preferences change
    final preferences = context.watch<Preferences>();
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
       // dark theme, and the choice between the two
      darkTheme: ThemeData(colorSchemeSeed: Colors.indigo, brightness: Brightness.dark),
      themeMode: preferences.sombre ? ThemeMode.dark : ThemeMode.light,
       // builder: wraps TOUS around the screens; here, the text size
      builder: (context, child) => MediaQuery(
        data: MediaQuery.of(context).copyWith(textScaler: TextScaler.linear(preferences.taille / 16)),
        child: child!,
      ),
      home: const PageEtatPartage(),
    );
  }
}

class PageEtatPartage extends StatefulWidget {
  const PageEtatPartage({super.key});

  @override
  State<PageEtatPartage> createState() => _PageEtatPartageState();
