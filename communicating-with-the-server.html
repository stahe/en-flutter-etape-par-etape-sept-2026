// lib/ex22_http/main.dart - an initial request HTTP to the server
// -------------------------------------------------------------------------
// The server JSON RdvMedecins must be running (port 8080). It is asked to
// a CAPTCHA: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//    - http.get(uri): the function from the http package that sends a request
//      GET. It returns a Future containing the response (http.Response): its
//      status code (statusCode), its headers, and its body;
//    - jsonDecode (text): the body JSON -> Dart objects (Map, List...);
//    - the server address depends on where the application is running
//      (Android emulator, phone, browser, etc.): see common/config.dart;
//    - three states to display: in progress, successful, error;
//    - the CAPTCHA image is from SVG: the flutter_svg package knows how to render it.
// =========================================================================

import 'dart:convert';

import 'package:flutter/material.dart';
import 'package:flutter_svg/flutter_svg.dart';
import 'package:http/http.dart' as http;

import '../commun/config.dart';

void main() {
  runApp(
    MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const PageCaptcha(),
    ),
  );
}

class PageCaptcha extends StatefulWidget {
  const PageCaptcha({super.key});

  @override
  State<PageCaptcha> createState() => _PageCaptchaState();
}

class _PageCaptchaState extends State<PageCaptcha> {
   // what GET returns from /api/captcha: a record
  ({String id, String svg})? _captcha;
  bool _enCours = true;
  String? _erreur;
  int _nbRequetes = 0;
  int _numeroDemande = 0; // the number of the last request

  @override
  void initState() {
    super.initState();
    _charger(); // the first request, as soon as the screen appears
  }

  Future<void> _charger() async {
    final numero = ++_numeroDemande;
    setState(() {
      _enCours = true;
      _erreur = null;
    });
    try {
      final reponse = await http.get(Uri.parse('${Config.urlServeur}/api/captcha'));
       // http.get does not throw an exception for an error response (4xx,
       // 5xx): the status code is checked
      if (reponse.statusCode != 200) {
        throw Exception('le serveur a répondu ${reponse.statusCode} ${reponse.reasonPhrase}');
      }
      final json = jsonDecode(reponse.body) as Map<String, dynamic>;
       // of the response to a request DÉPASSÉE (a new request was sent in the meantime), or
       // arrived after the screen closed: ignore it
      if (!mounted || numero != _numeroDemande) return;
      setState(() {
        _captcha = (id: json['id'] as String, svg: json['svg'] as String);
        _nbRequetes++;
        _enCours = false;
      });
    } catch (e) {
       // no response at all (server down, incorrect address...), or error
      if (!mounted || numero != _numeroDemande) return;
      setState(() {
        _erreur = '$e';
        _captcha = null;
        _enCours = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    final captcha = _captcha;
    return Scaffold(
      appBar: AppBar(title: const Text('Exemple 22 - une première requête HTTP')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          Text(
            'Serveur : ${Config.urlServeur.isEmpty ? '(même origine)' : Config.urlServeur}',
            style: const TextStyle(color: Colors.grey),
          ),
          const SizedBox(height: 12),
          if (_enCours) const LinearProgressIndicator(),
          if (_erreur != null)
            Card(
              color: Colors.red.shade50,
              child: Padding(
                padding: const EdgeInsets.all(12),
                child: Text(
                  "Impossible d'obtenir un captcha : $_erreur.\n"
                  'Le serveur JSON RdvMedecins est-il lancé (port 8080) ?',
                ),
              ),
            ),
          if (captcha != null && !_enCours) ...[
             // SvgPicture.string: draw a given image SVG as text
            Container(
              padding: const EdgeInsets.all(4),
              decoration: BoxDecoration(
                border: Border.all(color: Colors.grey.shade300),
                borderRadius: BorderRadius.circular(6),
              ),
              child: SvgPicture.string(captcha.svg, height: 60),
            ),
            const SizedBox(height: 8),
            Text('id : ${captcha.id}', style: const TextStyle(fontFamily: 'RobotoMono', fontSize: 11)),
          ],
          const SizedBox(height: 12),
          Align(
            alignment: Alignment.centerLeft,
            child: FilledButton(onPressed: _enCours ? null : _charger, child: const Text('Autre image')),
          ),
          const SizedBox(height: 8),
          Text('$_nbRequetes requête(s) réussie(s)', style: const TextStyle(color: Colors.grey)),
        ],
      ),
    );
  }
}

    // =========================================================================
// lib/commun/client_http.dart - a client HTTP that handles COOKIES
// -------------------------------------------------------------------------
// The server RdvMedecins places the token JWT in a COOKIE (httpOnly).
// In a browser, it is the browser that stores cookies and
// sends them back with every request: the application code has nothing to do with it.
// On a phone, there is no PAS browser: it is up to the application to
// fulfill this role, that is:
//    1. read the "Set-Cookie" header in each response and store the cookies;
//    2. send them back in the “Cookie” header of each request.
// This client (HTTP) does this; in ENVELOPPANT, the client in the HTTP package
// (http.BaseClient: simply call the `send` method). The cookies are
// also stored on the device (shared_preferences): you remain logged in
// even when you close and then reopen the app.
// On the web, it does nothing more than a regular client: the
// browser prevents the app from reading or writing these headers.
// =========================================================================

import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:http/http.dart' as http;
import 'package:shared_preferences/shared_preferences.dart';

class ClientAvecCookies extends http.BaseClient {
  static const _cle = 'rdvmedecins.cookies';

  final http.Client _interne = http.Client();
  final SharedPreferences? _stockage;
   // cookie name -> its value
  final Map<String, String> _cookies = {};

  ClientAvecCookies([this._stockage]) {
     // cookies stored from a previous session
    final memorises = _stockage?.getString(_cle);
    if (memorises != null) {
      _cookies.addAll((jsonDecode(memorises) as Map<String, Object?>).cast<String, String>());
    }
  }

  @override
  Future<http.StreamedResponse> send(http.BaseRequest requete) async {
    if (!kIsWeb && _cookies.isNotEmpty) {
       // "rdvmedecins_jwt=eyJhbGci...; other=value"
      requete.headers['cookie'] = [for (final MapEntry(:key, :value) in _cookies.entries) '$key=$value'].join('; ');
    }
    final reponse = await _interne.send(requete);
    final setCookie = reponse.headers['set-cookie'];
    if (!kIsWeb && setCookie != null) {
      _memoriser(setCookie);
    }
    return reponse;
  }

   // Set-Cookie: rdvmedecins_jwt=eyJ...; Max-Age=7200; Path=/api; Expires=Tue, Sep 29, 2026 ...; HttpOnly
   // (multiple cookies are separated by commas). We only keep the
   // name and value of each cookie; an empty value (logout) deletes it
  void _memoriser(String setCookie) {
     // a name followed by "=", at the beginning of the header or after a comma
    for (final m in RegExp(r'(?:^|,)\s*([^=;,\s]+)=([^;]*)').allMatches(setCookie)) {
      final nom = m[1]!;
      final valeur = m[2]!.trim();
      if (valeur.isEmpty) {
        _cookies.remove(nom);
      } else {
        _cookies[nom] = valeur;
      }
    }
    _stockage?.setString(_cle, jsonEncode(_cookies));
  }

  @override
  void close() => _interne.close();
}
    • Line 25: The [http] package allows you to wrap your client: You inherit from [http.BaseClient], and all you need to do is implement the [send] method, through which all requests pass ([get], [post]...);
    • lines 28–31: the wrapped client, the device’s storage (optional: [?]), and the cookies (name → value);
    • lines 33–37: upon creation, the cookies stored during a previous use of the app. This keeps you logged in when you close and then reopen the app: on a phone, this is equivalent to a browser retaining its cookies from one visit to the next;
    • lines 42–52: Each request receives the header [Cookie] (“rdvmedecins_jwt=eyJhbGci...; other=value”), then the [Set-Cookie] header from the response is read. [kIsWeb]: On the web, the app does nothing more than a regular client, because the browser prevents the app from reading or writing these headers (it handles them itself);
    • lines 58–69: the [Set-Cookie] header. Several cookies are separated by commas, but the date [Expires] itself contains a comma (“Tue, 29 Sep 2026”): the regular expression matches only a name followed by [=], either at the beginning of the header or after a comma. Only the name and value of each cookie are retained; an empty value (logout: the server sends [rdvmedecins_jwt=; Expires=Thu, 01 Jan 1970...]) deletes the cookie. [_stockage?.setString(...)]: The call is made only if [_stockage] is not [null].

    More comprehensive cookie management (domains, paths, expiration dates) is available in packages ([cookie_jar], with the client [dio]); our implementation, consisting of just a few lines, is sufficient for the UN server. Security note: The token is stored in plain text on the device here. A production application would use the [flutter_secure_storage] package (see Example 20).

// lib/ex23_connexion/commande.dart - a ACTION, its status, its result
// -------------------------------------------------------------------------
// An asynchronous action (log in, save...) has a status that
// the screen must display: Is it EN COURS (button disabled, indicator)?
//end with a ERREUR (message)? This is the "Command" template
// from the Flutter architecture guide, stripped down to the essentials:
//
//      final connection = Command();
//      connexion.executer(() => seConnecter(...   )); // triggers the action
//      connexion.enCours, connexion.err            eur // its state
//
// A command is a ChangeNotifier: the screen listens for it (ListenableBuilder)
// and redraws itself when the command starts and when it ends.
// =========================================================================

import 'package:flutter/foundation.dart';

class Commande extends ChangeNotifier {
  bool _enCours = false;
  Object? _erreur;

  bool get enCours => _enCours;
  Object? get erreur => _erreur;

   // executes the action, if another one isn’t already in progress
  Future<void> executer(Future<void> Function() action) async {
    if (_enCours) return; // prevents double-sending (double-tapping the button)
    _enCours = true;
    _erreur = null;
    notifyListeners();
    try {
      await action();
    } catch (e) {
      _erreur = e; // the error is GARDÉE, to be displayed
    } finally {
       // Finally: executed in TOUS in all cases, whether successful or an error
      _enCours = false;
      notifyListeners();
    }
  }

  void effacerErreur() {
    _erreur = null;
    notifyListeners();
  }
}

    An asynchronous action (log in, save, etc.) has a status that the screen must display: is it in progress (button disabled)? Did it end with an error (message)? This is the “Command” model from the [Flutter] architecture guide, stripped down to the essentials. The [React] course used the “actions” from [React] 19 ([useActionState]) for this purpose.

    • Lines 27–39: Execute an action: if another action is in progress, do nothing (to prevent double-sending if the button is pressed twice); otherwise, the command changes to “in progress” and notifies its listeners; in the event of an exception, the error is logged for display; in all cases ([finally]), the command is no longer in progress.

// "implements Exception": an object that can be thrown and caught
class ErreurApi implements Exception {
  final int statusCode; // 0: the server did not respond
  final String cle; // e.g., 'ERRORS.LOGIN_TAKEN'
  final Map<String, String> params;
  final Map<String, String> champs;

  const ErreurApi(this.statusCode, this.cle, {this.params = const {}, this.champs = const {}});

   // an error response (4xx, 5xx) -> a ErreurApi
  factory ErreurApi.depuisReponse(http.Response reponse) {
    Object? corps;
    try {
      corps = jsonDecode(utf8.decode(reponse.bodyBytes));
    } on FormatException {
      corps = null; // not JSON: this response did not come from our server
    }
     // "filtering" (pattern matching) of a Map: if the body has a key
     // "key" of type String, it is stored in the variable key
    if (corps case {'cle': final String cle}) {
      final json = corps as Map<String, Object?>;
      return ErreurApi(reponse.statusCode, cle, params: _chaines(json['params']), champs: _chaines(json['champs']));
    }
     // No key: the proxy or the network responded instead
    return ErreurApi(
      reponse.statusCode,
      reponse.statusCode >= 500 ? 'ERRORS.SERVER_UNREACHABLE' : 'ERRORS.SERVER_ERROR',
    );
  }

   // an object JSON { "name": "ERRORS..." } -> Map<String, String> (empty otherwise)
  static Map<String, String> _chaines(Object? json) =>
      json is Map<String, Object?> ? {for (final e in json.entries) e.key: '${e.value}'} : const {};

  @override
  String toString() => '$statusCode $cle';
}

// All requests go through the `requete` method: this is where
// a problem like COMMUN would be handled for all of them (see the case study and the
// 401 response). Used in Examples 24 and 25.

import 'dart:convert';

import 'package:http/http.dart' as http;

import 'config.dart';
import 'erreur_api.dart';

class Api {
  final http.Client _client;

  Api(this._client);

  Future<Object?> requete(String methode, String url, [Object? corps]) async {
    final requete = http.Request(methode, Uri.parse('${Config.urlServeur}/api$url'));
    if (corps != null) {
       // a body JSON (the server rejects a POST that is not a body)
      requete.headers['content-type'] = 'application/json';
      requete.body = jsonEncode(corps);
    }
    final http.Response reponse;
    try {
       // send sends the request; fromStream waits for the complete response body
      reponse = await http.Response.fromStream(await _client.send(requete));
    } catch (_) {
       // AUCUNE response (network issue, server down, incorrect address)
      throw const ErreurApi(0, 'ERRORS.SERVER_UNREACHABLE');
    }
    if (reponse.statusCode >= 400) {
      throw ErreurApi.depuisReponse(reponse);
    }
    return reponse.bodyBytes.isEmpty ? null : jsonDecode(utf8.decode(reponse.bodyBytes));
  }

  Future<Object?> get(String url) => requete('GET', url);
   // a POST always has a body JSON: {} by default
  Future<Object?> post(String url, [Object corps = const {}]) => requete('POST', url, corps);
  Future<Object?> put(String url, Object corps) => requete('PUT', url, corps);
  Future<Object?> delete(String url, [Object? corps]) => requete('DELETE', url, corps);
}

String _deuxChiffres(int n) => n.toString().padLeft(2, '0');
String formatJour(DateTime d) => '${d.year}-${_deuxChiffres(d.month)}-${_deuxChiffres(d.day)}';

class AgendaModele extends ChangeNotifier {
  final RdvMedecins _rdvmedecins;

  AgendaModele(this._rdvmedecins);

   // ---- status ----
  Utilisateur? utilisateur;
  List<Personne> medecins = [];
  List<Personne> clients = [];
  int? idMedecin;
  String jour = formatJour(DateTime.now().add(const Duration(days: 1))); // tomorrow
  int? idClient;
  Agenda? agenda;
  bool chargement = false;
  ({bool succes, String texte})? message;

  final Map<String, Agenda> _cache = {};
  int _numeroLecture = 0;

   // ---- calculated values ----
  bool get connecte => utilisateur != null;
  bool get gestionnaire => connecte && utilisateur!.role != Role.user;

   // ---- actions ----
  Future<void> changerUtilisateur(Utilisateur? u) async {
    utilisateur = u;
    medecins = [];
    clients = [];
    agenda = null;
    idMedecin = null;
    _cache.clear(); // permissions have changed: forget everything
    notifyListeners();
    if (u == null) return;
    medecins = await _rdvmedecins.getMedecins();
    if (gestionnaire) clients = await _rdvmedecins.getClients(); // a USER would receive 403
    notifyListeners();
  }

  void choisirMedecin(int id) {
    idMedecin = id;
    message = null;
    _lireAgenda();
  }

  void choisirJour(DateTime d) {
    jour = formatJour(d);
    message = null;
    _lireAgenda();
  }

  void choisirClient(int? id) {
    idClient = id;
    notifyListeners();
  }

  Future<void> reserver(CreneauAgenda c) =>
      _agir(() => _rdvmedecins.reserver(jour, c.id, gestionnaire ? idClient : null), 'Créneau ${c.horaire} réservé.');

  Future<void> annuler(CreneauAgenda c) =>
      _agir(() => _rdvmedecins.annuler(c.idRv!), 'Rendez-vous de ${c.horaire} annulé.');

   // an action, then TOUJOURS a review of the agenda: SERVEUR is the definitive version
  Future<void> _agir(Future<void> Function() action, String succes) async {
    try {
      await action();
      message = (succes: true, texte: succes);
    } on ErreurApi catch (e) {
      message = (succes: false, texte: '$e');
    }
    await _lireAgenda();
  }

  Future<void> _lireAgenda() async {
    final id = idMedecin;
    if (id == null) return;
    final cle = '$id/$jour';
    final numero = ++_numeroLecture;
     // the cache first (or nothing), then the read
    agenda = _cache[cle];
    chargement = true;
    notifyListeners();
    try {
      final lu = await _rdvmedecins.getAgenda(id, jour);
      if (numero != _numeroLecture) return; // a more recent read was initiated
      agenda = _cache[cle] = lu;
    } on ErreurApi catch (e) {
      if (numero != _numeroLecture) return;
      agenda = null;
      message = (succes: false, texte: '$e');
    }
    chargement = false;
    notifyListeners();
  }
}

    setState(() {
      _idEdite = null;
      _version = 0;
      _titre = 'Mr';
      for (final c in _textes.values) {
        c.clear();
      }
      _effacerMessages();

    setState(() {
      _idEdite = c.id;
       // the VERSION read from the record is sent back with the modification: if someone
       // has modified the record in the meantime, the server will respond with a 409 error: CONCURRENT_UPDATE
      _version = c.version;
      _titre = c.titre;
      _textes[Champ.prenom]!.text = c.prenom;
      _textes[Champ.nom]!.text = c.nom;
      _textes[Champ.login]!.text = c.login ?? '';
      _textes[Champ.password]!.clear();
      _effacerMessages();
    });
  }

    setState(_effacerMessages);
    final saisie = {
      'titre': _titre,
      for (final MapEntry(key: champ, value: c) in _textes.entries) champ.name: c.text,
      'version': _version,

      final client = Personne.depuisJson(
        _idEdite == null ? await _api.post('/clients', saisie) : await _api.put('/clients/$_idEdite', saisie),

      );

        _erreurPage = e; // the general message
        _erreurs = e.champs; // the error for each field
      });

      await _api.delete('/clients/${c.id}');

    } on ErreurApi catch (e) {
      setState(() => _erreurPage = e);
    }
